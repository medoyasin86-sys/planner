#!/usr/bin/env python3 # -*- coding: utf-8 -*- """ MYH Planner PDF Generator - GUARANTEED TO WORK Uses Playwright with real Chromium engine for pixel-perfect PDF output No browser memory issues - runs server-side """ import asyncio import sys from pathlib import Path from datetime import datetime, timedelta async def generate_planner(): """Generate complete planner PDF using Playwright""" try: from playwright.async_api import async_playwright except ImportError: print("❌ Playwright not installed. Run: pip install playwright") print("Then run: python -m playwright install chromium") return # Paths html_file = Path("MYH_Planner_2026_2027_Export_Dashboard.html").absolute() output_dir = Path("output_pdfs") output_dir.mkdir(exist_ok=True) if not html_file.exists(): print(f"❌ HTML file not found: {html_file}") print("Please ensure the HTML file is in the same directory") return print(f"📄 Loading HTML: {html_file}") print(f"📁 Output directory: {output_dir}") async with async_playwright() as p: # Launch Chromium with memory optimization print("🚀 Launching Chromium...") browser = await p.chromium.launch( headless=True, args=[ '--no-sandbox', '--disable-dev-shm-usage', '--disable-gpu', '--disable-web-security', '--disable-features=IsolateOrigins,site-per-process', '--memory-pressure-off', '--js-flags=--max-old-space-size=4096' ] ) # Create page with exact planner dimensions context = await browser.new_context( viewport={'width': 1024, 'height': 768}, device_scale_factor=1 # Use 1 for memory efficiency, 2 for quality ) page = await context.new_page() # Load HTML file print("⏳ Loading page content...") await page.goto(f"file://{html_file}") await page.wait_for_load_state('networkidle') # Wait for fonts and scripts to initialize print("⏳ Waiting for fonts and scripts...") await asyncio.sleep(3) # Wait for fonts to be ready try: await page.evaluate("document.fonts.ready") print("✅ Fonts loaded") except: print("⚠️ Font loading timeout, continuing...") # Month definitions months = [ (2026, 6, 30), (2026, 7, 31), (2026, 8, 31), (2026, 9, 30), (2026, 10, 31), (2026, 11, 30), (2026, 12, 31), (2027, 1, 31), (2027, 2, 28), (2027, 3, 31), (2027, 4, 30), (2027, 5, 31), (2027, 6, 30), (2027, 7, 31), (2027, 8, 31), (2027, 9, 30), (2027, 10, 31), (2027, 11, 30), (2027, 12, 31), ] months_ar = ["يناير", "فبراير", "مارس", "أبريل", "مايو", "يونيو", "يوليو", "أغسطس", "سبتمبر", "أكتوبر", "نوفمبر", "ديسمبر"] total_months = len(months) # Generate each month as separate PDF for month_idx, (year, month, days) in enumerate(months): month_name = months_ar[month - 1] print(f" 📅 [{month_idx+1}/{total_months}] Generating {month_name} {year} ({days} days)...") month_pdfs = [] # For first month, add yearly overview page if month_idx == 0: print(" 📊 Adding yearly overview...") await page.evaluate("switchView('yearly')") await asyncio.sleep(1) yearly_pdf = await page.pdf( width="1024px", height="768px", print_background=True, prefer_css_page_size=True ) month_pdfs.append(yearly_pdf) print(" ✅ Yearly overview added") # Generate daily pages for day in range(1, days + 1): date_str = f"{year}-{month:02d}-{day:02d}" # Navigate to date await page.evaluate(f"goToDate('{date_str}')") await asyncio.sleep(0.3) # Work mode page await page.evaluate("setMode('work')") await asyncio.sleep(0.2) work_pdf = await page.pdf( width="1024px", height="768px", print_background=True, prefer_css_page_size=True ) month_pdfs.append(work_pdf) # Personal mode page await page.evaluate("setMode('personal')") await asyncio.sleep(0.2) personal_pdf = await page.pdf( width="1024px", height="768px", print_background=True, prefer_css_page_size=True ) month_pdfs.append(personal_pdf) # Progress if day % 10 == 0 or day == days: print(f" Progress: {day}/{days} days") # Memory cleanup every 5 days if day % 5 == 0: await page.evaluate("if(window.gc) window.gc()") # Save month PDF month_file = output_dir / f"MYH_{year}_{month:02d}_{month_name}.pdf" with open(month_file, 'wb') as f: for pdf_data in month_pdfs: f.write(pdf_data) file_size = month_file.stat().st_size / (1024 * 1024) print(f" ✅ Saved: {month_file.name} ({file_size:.1f} MB, {len(month_pdfs)} pages)") # Clear month_pdfs from memory month_pdfs.clear() # Pause between months for memory cleanup await asyncio.sleep(1) # Merge all PDFs using PyPDF2 print(" 📚 Merging all months into complete PDF...") try: from PyPDF2 import PdfMerger merger = PdfMerger() total_pages = 0 for year, month, _ in months: month_name = months_ar[month - 1] month_file = output_dir / f"MYH_{year}_{month:02d}_{month_name}.pdf" if month_file.exists(): merger.append(str(month_file)) # Count pages from PyPDF2 import PdfReader reader = PdfReader(str(month_file)) total_pages += len(reader.pages) complete_file = output_dir / "MYH_Planner_2026_2027_Complete.pdf" merger.write(str(complete_file)) merger.close() complete_size = complete_file.stat().st_size / (1024 * 1024) print(f" 🎉 SUCCESS!") print(f"📄 Complete PDF: {complete_file}") print(f"📊 Total pages: {total_pages}") print(f"💾 File size: {complete_size:.1f} MB") print(f"📁 Month files: {total_months} files in {output_dir}") except ImportError: print(" ⚠️ PyPDF2 not available for merging") print("📁 Individual month PDFs are saved in:", output_dir) await browser.close() print(" ✅ Done! Check your output_pdfs folder.") if __name__ == "__main__": asyncio.run(generate_planner())
